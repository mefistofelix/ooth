package poll

import "syscall"

// Keep each AcceptEx completion local to its process. Go's event-based path
// already sets the event's low bit to suppress packets to an external IOCP.
func (fd *FD) waitSharedIO(o *operation) error {
	for {
		status, err := syscall.WaitForSingleObject(o.o.HEvent&^1, 20)
		if err != nil {
			return err
		}
		if status == syscall.WAIT_OBJECT_0 {
			return nil
		}
		if err := fd.pd.prepare(int(o.mode), fd.isFile); err != nil {
			if cancelErr := syscall.CancelIoEx(fd.Sysfd, &o.o); cancelErr != nil && cancelErr != syscall.ERROR_NOT_FOUND {
				panic(cancelErr)
			}
			// The kernel must release the pinned buffers before execIO returns.
			syscall.WaitForSingleObject(o.o.HEvent&^1, syscall.INFINITE)
			return err
		}
	}
}
